package main

import (
	"net/http"

	"github.com/gin-gonic/gin"
)

type Booking struct {
	Name      string `form:"name"`
	Email     string `form:"email"`
	CheckIn   string `form:"checkin"`
	CheckOut  string `form:"checkout"`
	RoomType  string `form:"roomtype"`
	Guests    string `form:"guests"`
}

func main() {
	r := gin.Default()
	r.LoadHTMLGlob("templates/*")
	r.Static("/static", "./static")

	r.GET("/", func(c *gin.Context) {
		c.HTML(http.StatusOK, "index.html", nil)
	})

	r.GET("/rooms", func(c *gin.Context) {
		c.HTML(http.StatusOK, "rooms.html", nil)
	})

	r.GET("/booking", func(c *gin.Context) {
		c.HTML(http.StatusOK, "booking.html", nil)
	})

	r.POST("/booking", func(c *gin.Context) {
		var booking Booking
		if err := c.ShouldBind(&booking); err != nil {
			c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
			return
		}

		c.HTML(http.StatusOK, "success.html", gin.H{
			"name": booking.Name,
		})
	})

	r.Run(":8080")
}
